function(dv_add_test name)
    add_executable(${name} unit/${name}.cpp)
    target_link_libraries(${name} PRIVATE dv_core dv_settings dv_localization dv_project dv_media dv_timeline dv_playback dv_export dv_ai Qt6::Test)
    target_compile_definitions(${name} PRIVATE DV_TEST_FFMPEG="${DV_FFMPEG_DIR}/bin/ffmpeg.exe")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;DV_NO_AUDIO=1")
endfunction()
